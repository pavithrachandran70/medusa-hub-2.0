# Story 01-medusa-appli: Database schema and wishlist service layer

**Repo:** `medusa-appli`

## Context

Establish the foundational data model and business logic for wishlist persistence. This story creates the database table with foreign key constraints to customer and product, and implements a service layer for add/list/remove operations with validation and idempotency enforcement.

## Acceptance Criteria

1. Customers can add a product to their wishlist (R1: validated via service method).
2. Adding the same product multiple times does not create duplicate entries (R1: unique constraint enforced, service returns existing entry on duplicate).
3. Wishlist data persists when the customer logs out and logs in again (R4: data stored in database table with foreign key to customer account).
4. Customers can remove products from their wishlist (R3: service method deletes row).
5. Only authenticated customers can manage their wishlist (enforced in Story 2 API layer; this story provides the validated service methods).

#### Function: addToWishlist

```typescript
async addToWishlist(
  customerId: string,
  productId: string
): Promise<Wishlist> {
  // Validate customer exists
  const customer = await this.customerModuleService_.retrieveCustomer(customerId)
  if (!customer) {
    throw new Error(`Customer with id ${customerId} not found`)
  }

  // Validate product exists
  const product = await this.productModuleService_.retrieveProduct(productId)
  if (!product) {
    throw new Error(`Product with id ${productId} not found`)
  }

  // Attempt to create wishlist entry
  try {
    const wishlist = this.manager_.create(Wishlist, {
      id: this.generateId(), // verify ID generation pattern in existing services
      customer: customerId,
      product: productId,
    })

    await this.manager_.persistAndFlush(wishlist)
    return wishlist
  } catch (error) {
    // If unique constraint violation, query and return existing entry (idempotent)
    if (error.code === "23505") { // PostgreSQL unique violation code
      const existing = await this.manager_.findOne(Wishlist, {
        customer: customerId,
        product: productId,
      })
      if (existing) {
        return existing
      }
    }
    throw error
  }
}
```

**Key details:**
- Validates customer and product exist per AD-3 (business rule enforcement).
- Creates new `Wishlist` entity with foreign key references.
- **Idempotency per AD-6:** catches unique constraint violation (PostgreSQL error code `23505`), queries existing row, returns it without error. If constraint violation but no existing row found (race condition), re-throws error.
- Verify `generateId()` method or use pattern from existing services (may be `uuidv4()` or framework utility).
- Verify `persistAndFlush()` vs. separate `persist()` + `flush()` in existing code.

**Open question:** Verify exact error code for unique constraint violation in MikroORM + PostgreSQL. May need to check `error.constraint` or `error.detail` instead of `error.code`.

```typescript
async removeFromWishlist(
  customerId: string,
  productId: string
): Promise<void> {
  const wishlist = await this.manager_.findOne(Wishlist, {
    customer: customerId,
    product: productId,
  })

  if (wishlist) {
    await this.manager_.removeAndFlush(wishlist)
  }

  // No error if row does not exist per AD-3
}
```

**Key details:**
- Deletes the wishlist row matching both `customer` and `product`.
- No error thrown if row does not exist (idempotent delete).
- Uses `removeAndFlush()` per MikroORM patterns (verify in existing services).

```typescript
import { medusaIntegrationTestRunner } from "@medusajs/test-utils"
import {
  createAdminUser,
  generatePublishableKey,
  generateStoreHeaders,
} from "../../../../helpers/create-admin-user"
import { Modules } from "@medusajs/utils"

const env = {}
const adminHeaders = { headers: { "x-medusa-access-token": "test_token" } }

medusaIntegrationTestRunner({
  env,
  testSuite: ({ dbConnection, getContainer, api }) => {
    let appContainer
    let storeHeaders

    beforeAll(async () => {
      appContainer = getContainer()
      await createAdminUser(dbConnection, adminHeaders, appContainer)
      const publishableKey = await generatePublishableKey(appContainer)
      storeHeaders = generateStoreHeaders({ publishableKey })
    })

    describe("Wishlist Service Layer Tests (Story 1)", () => {
      describe("Service idempotency", () => {
        it("should return the same wishlist item when adding the same product twice", async () => {
          const container = getContainer()
          const wishlistService = container.resolve("wishlistService")
          const customerModule = container.resolve(Modules.CUSTOMER)
          const productModule = container.resolve(Modules.PRODUCT)

          // Create test customer and product
          const customer = await customerModule.createCustomers({
            email: "test@example.com",
          })
          const product = await productModule.createProducts({
            title: "Test Product",
            status: "published",
          })

          // First add
          const firstAdd = await wishlistService.addToWishlist(
            customer.id,
            product.id
          )

          // Second add (duplicate)
          const secondAdd = await wishlistService.addToWishlist(
            customer.id,
            product.id
          )

          // Verify same ID returned
          expect(secondAdd.id).toEqual(firstAdd.id)

          // Verify database contains exactly one row
          const dbConnection = container.resolve("dbConnection")
          const result = await dbConnection.raw(
            `SELECT COUNT(*) FROM wishlist WHERE customer_id = ? AND product_id = ?`,
            [customer.id, product.id]
          )
          expect(parseInt(result.rows[0].count)).toEqual(1)
        })
      })

      describe("Cascade delete behavior", () => {
        it("should automatically remove wishlist item when product is deleted", async () => {
          const container = getContainer()
          const wishlistService = container.resolve("wishlistService")
          const customerModule = container.resolve(Modules.CUSTOMER)
          const productModule = container.resolve(Modules.PRODUCT)

          // Create test customer and product
          const customer = await customerModule.createCustomers({
            email: "cascade-test@example.com",
          })
          const product = await productModule.createProducts({
            title: "Cascade Test Product",
            status: "published",
          })

          // Add product to wishlist
          await wishlistService.addToWishlist(customer.id, product.id)

          // Delete the product
          await productModule.deleteProducts([product.id])

          // Verify wishlist row is automatically removed
          const items = await wishlistService.listWishlistItems(customer.id)
          expect(items).toHaveLength(0)
        })
      })

      describe("N+1 query prevention", () => {
        it("should eagerly load product relation and avoid N+1 queries", async () => {
          const container = getContainer()
          const wishlistService = container.resolve("wishlistService")
          const customerModule = container.resolve(Modules.CUSTOMER)
          const productModule = container.resolve(Modules.PRODUCT)

          // Create test customer
          const customer = await customerModule.createCustomers({
            email: "n-plus-one-test@example.com",
          })

          // Create two products
          const product1 = await productModule.createProducts({
            title: "Product 1",
            thumbnail: "thumb1.jpg",
            status: "published",
          })
          const product2 = await productModule.createProducts({
            title: "Product 2",
            thumbnail: "thumb2.jpg",
            status: "published",
          })

          // Add both to wishlist
          await wishlistService.addToWishlist(customer.id, product1.id)
          await wishlistService.addToWishlist(customer.id, product2.id)

          // Enable query logging
          const dbConnection = container.resolve("dbConnection")
          const originalQuery = dbConnection.query
          const queries: string[] = []
          dbConnection.query = function (...args: any[]) {
            queries.push(args[0])
            return originalQuery.apply(this, args)
          }

          // List wishlist items
          const items = await wishlistService.listWishlistItems(customer.id)

          // Restore original query function
          dbConnection.query = originalQuery

          // Verify both items returned with product relation populated
          expect(items).toHaveLength(2)
          expect(items[0].product).toBeDefined()
          expect(items[0].product.title).toBeDefined()
          expect(items[0].product.thumbnail).toBeDefined()
          expect(items[1].product).toBeDefined()
          expect(items[1].product.title).toBeDefined()
          expect(items[1].product.thumbnail).toBeDefined()

          // Verify no N+1: should be 1 query with JOIN, not 1 + N separate queries
          const selectQueries = queries.filter((q) =>
            q.toLowerCase().includes("select")
          )
          expect(selectQueries.length).toBeLessThanOrEqual(2) // Allow for initial query + possible metadata query
        })
      })
    })
  },
})
```
